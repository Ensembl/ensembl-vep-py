197 @classmethod 

198 def subsumes_so_parent_drift(cls, ruleset: FactRuleSet) -> list[str]: 

199 """Report where the declared ``subsumes`` graph disagrees with the SO ``so_supported_parents``. 

200 

201 These are two genuinely different relations that happen to overlap a lot: ``subsumes`` is 

202 VEP's empirical suppression behaviour (term A, if it fires, hides term B), while 

203 ``so_supported_parents`` is the Sequence Ontology's ``is_a`` parent graph (term A *is a kind 

204 of* term B), restricted to parents also represented in this catalogue. A rule can subsume a 

205 term it is not an SO child of (e.g. suppressing a sibling), and a rule can have an SO parent it 

206 does not need to suppress (the parent's own ``when`` may already be false whenever the child's

277# Tier 2: per-rule enumeration and per-pair propositional search. 

278# 

279# A rule's atom *support* is its formula's own atoms, extended by the requires-closure (every 

280# atom that some support atom's declared `requires` would force, transitively) and by a small 

281# hand-declared exclusivity/equivalence axiom set covering relationships the fact layer's 

282# `requires` mechanism does not itself express (e.g. "is_insertion and is_deletion cannot both 

283# hold" is true by construction of the two fact functions, but neither fact declares the other in 

284# its `requires`, since `requires` only expresses "if X is False, so am I", not general mutual 

285# exclusion). Enumerating every assignment over that support and discarding the ones the axioms 

286# rule out gives every *possible* combination of fact values a real Context could produce, which 

287# is exactly what "is this rule ever true" needs to be sound. See the module docstring for why 

288# this -- not equality saturation -- is the right tool for that question. 

289# -------------------------------------------------------------------------- 

290

293 # alternate_allele_length cannot both hold. 

294 ("is_insertion", "is_deletion"), 

295 # is_upstream/is_downstream both explicitly return False whenever the variant overlaps the 

296 # transcript at all (facts/position.py).

564 satisfying_assignment_count += 1 

565 return cls( 

566 term=rule.term, 

567 support=space.support, 

568 valid_assignment_count=valid_assignment_count, 

569 satisfying_assignment_count=satisfying_assignment_count, 

570 )

572 @classmethod 

573 def unreachable_in(cls, ruleset: FactRuleSet) -> list[RuleReachability]: 

574 """Return every rule whose ``when`` formula is unsatisfiable under the Tier 2 axioms. 

575 

576 An unreachable rule is dead code: it can never contribute a term to any output, which -- unlike 

577 the CONTRADICTORY/REDUNDANT findings below -- is a genuine defect rather than informational 

578 context, so callers (the validation script, the test suite) should treat a non-empty result as 

579 a build-breaking error. 

580 

581 Args: 

582 ruleset (FactRuleSet): The rule set to check. 

583 

584 Returns: 

585 list[RuleReachability]: The unreachable rules' results (``is_reachable`` False for all of 

586 them), or an empty list if every rule is reachable. 

587 """ 

588 return [result for rule in ruleset.rules if not (result := cls.for_rule(rule)).is_reachable]

591@frozen(kw_only=True, auto_attribs=True) 

592class RulePairFinding: 

593 """Record one pairwise rule-validation finding. 

594 

595 Attributes: 

596 kind (str): Either ``"CONTRADICTORY"`` (the two rules' formulas can never both hold) or 

597 ``"REDUNDANT"`` (rule ``term_a`` subsumes ``term_b`` *and* ``term_a``'s formula 

598 logically entails ``term_b``'s, confirming the subsumption is logically grounded). 

599 Both kinds are informational governance signal, not build-breaking errors -- see 

600 RulePairFinding.contradictory_in/RulePairFinding.redundant_in. 

601 term_a (str): The first rule's term. 

602 term_b (str): The second rule's term. 

603 message (str): Human-readable description. 

604 """ 

605 

606 kind: str 

607 term_a: str 

608 term_b: str 

609 message: str 

610 

611 @classmethod 

612 def contradictory_in( 

613 cls, ruleset: FactRuleSet, *, max_union_support: int = _MAX_UNION_SUPPORT_FOR_PAIR_ENUMERATION 

614 ) -> list[RulePairFinding]: 

615 """Find rule pairs whose formulas can never both hold, and that do not subsume each other. 

616 

617 This is informational rather than an error: two rules never co-firing is often simply how the 

618 ruleset is meant to work (e.g. ``stop_gained`` and ``synonymous_variant`` are naturally 

619 mutually exclusive), not a defect. It is useful context for a catalogue reviewer -- a pair 

620 that *should* be able to co-fire but this check finds contradictory would be worth a second 

621 look -- rather than something a script should fail the build over. 

622 

623 Args: 

624 ruleset (FactRuleSet): The rule set to check every pair of. 

625 max_union_support (int): Skip a pair whose combined atom support exceeds this size, so a 

626 future larger catalogue degrades to "fewer pairs checked" rather than a slow/unbounded 

627 enumeration. 

628 

629 Returns: 

630 list[RulePairFinding]: One ``"CONTRADICTORY"`` finding per such pair. 

631 """ 

632 findings: list[RulePairFinding] = [] 

633 rules = ruleset.rules 

634 spaces = {rule.term: FactAssignmentSpace.for_rule(rule) for rule in rules} 

635 

636 for index_a, rule_a in enumerate(rules): 

637 for index_b in range(index_a + 1, len(rules)): 

638 rule_b = rules[index_b] 

639 if rule_b.term in rule_a.subsumes or rule_a.term in rule_b.subsumes: 

640 continue 

641 union = spaces[rule_a.term].union(spaces[rule_b.term]) 

642 if len(union.support) > max_union_support: 

643 continue 

644 can_both_hold = union.satisfies(And(rule_a.when, rule_b.when)) 

645 if not can_both_hold: 

646 findings.append( 

647 RulePairFinding( 

648 kind="CONTRADICTORY", 

649 term_a=rule_a.term, 

650 term_b=rule_b.term, 

651 message=f"{rule_a.term} and {rule_b.term} never co-fire under the Tier 2 axioms", 

652 ) 

653 ) 

654 return findings

656 @classmethod 

657 def redundant_in( 

658 cls, ruleset: FactRuleSet, *, max_union_support: int = _MAX_UNION_SUPPORT_FOR_PAIR_ENUMERATION 

659 ) -> list[RulePairFinding]: 

660 """Find declared ``subsumes`` edges whose entailment is confirmed by the formulas themselves. 

661 

662 For each declared ``A subsumes B``, check whether ``A.when`` logically entails ``B.when`` over 

663 every valid assignment of their combined support (i.e. whenever A's formula could possibly be 

664 true, B's formula is necessarily true too). Where it holds, the subsumption is *logically 

665 redundant* in the sense that A firing already guarantees B's condition holds -- the ``subsumes`` 

666 declaration is provably sound, not merely asserted. Where it does not hold under this bounded 

667 axiom set, that is *not* reported as an error: most of this catalogue's subsumption edges rely 

668 on domain knowledge broader than the small declared axiom set here (e.g. "a deletion spanning 

669 the whole transcript necessarily also overlaps every intron and UTR" is true, but nothing in 

670 _EXCLUSIVITY_AXIOM_PAIRS encodes it), so a pair not appearing in this report is 

671 inconclusive, not suspect. 

672 

673 Args: 

674 ruleset (FactRuleSet): The rule set to check every declared ``subsumes`` edge of. 

675 max_union_support (int): Skip an edge whose combined atom support exceeds this size. 

676 

677 Returns: 

678 list[RulePairFinding]: One ``"REDUNDANT"`` finding per confirmed-entailed edge. 

679 """ 

680 findings: list[RulePairFinding] = [] 

681 spaces = {rule.term: FactAssignmentSpace.for_rule(rule) for rule in ruleset.rules} 

682 

683 for rule_a in ruleset.rules: 

684 for subsumed_term in rule_a.subsumes: 

685 rule_b = ruleset.by_term.get(subsumed_term) 

686 if rule_b is None: 

687 continue 

688 union = spaces[rule_a.term].union(spaces[rule_b.term]) 

689 if len(union.support) > max_union_support: 

690 continue 

691 counterexample = And(rule_a.when, Not(rule_b.when)) 

692 entailed = not union.satisfies(counterexample) 

693 if entailed: 

694 findings.append( 

695 RulePairFinding( 

696 kind="REDUNDANT", 

697 term_a=rule_a.term, 

698 term_b=subsumed_term, 

699 message=( 

700 f"{rule_a.term}'s when provably entails {subsumed_term}'s when -- the " 

701 "declared subsumption is logically sound, confirmed under the Tier 2 axioms" 

702 ), 

703 ) 

704 ) 

705 return findings

761 @classmethod 

762 def for_catalogue_entries( 

763 cls, entries: list[dict[str, Any]], baseline_entries: list[dict[str, Any]] 

764 ) -> ValidationReport: 

765 """Validate a candidate ``consequences`` list before it is ever written to disk. 

766 

767 Reuses ensembl.vep.annotators.consequence.rules.validation's structural (Tier 1) and enumeration 

768 (Tier 2) checks over a FactRuleSet built from the *candidate* entries (see 

769 _ruleset_from_entries) -- not the packaged catalogue, which is what makes this safe to 

770 call before a real term has been added or removed on disk. The two checks that hardcode a call 

771 to the packaged load_consequence_definitions() (report_missing_when_terms, 

772 report_subsumes_so_parent_drift) are intentionally not reused here, since they are defined in 

773 terms of the checked-in EXCLUDED_TERMS/catalogue rather than a candidate in-memory one; a 

774 needs-review, ``when``-less addition is instead surfaced separately, at add-time, by 

775 resolve_when's NEEDS REVIEW banner. 

776 

777 Unreachable-rule findings are diffed against ``baseline_entries`` (the catalogue before this 

778 add/remove was applied) rather than treated as build-breaking outright: the checked-in 

779 catalogue already carries one known, pre-existing dead rule (``feature_truncation``). Treating 

780 every unreachable rule as build-breaking here would permanently refuse every write against the 

781 real catalogue, for a reason unrelated to whatever term is being added or removed. A rule 

782 unreachable in *both* the baseline and the candidate is downgraded to an informational warning; 

783 one that only became unreachable in the candidate is a genuine regression introduced by this 

784 operation and stays build-breaking. 

785 

786 Args: 

787 entries (list[dict[str, Any]]): The candidate catalogue's ``consequences`` list. 

788 baseline_entries (list[dict[str, Any]]): The catalogue's ``consequences`` list *before* 

789 this operation, used only to distinguish pre-existing unreachable rules from new ones. 

790 

791 Returns: 

792 ValidationReport: errors/unreachable_rules non-empty means refuse to write; warnings are 

793 informational only (see ValidationReport.has_errors). 

794 """ 

795 ruleset = cls._ruleset_from_entries(entries) 

796 structural_errors, warnings = StructuralChecks.ruleset_problems(ruleset) 

797 errors = StructuralChecks.term_and_rank_uniqueness(entries) + structural_errors 

798 

799 baseline_ruleset = cls._ruleset_from_entries(baseline_entries) 

800 baseline_unreachable_terms = frozenset( 

801 result.term for result in RuleReachability.unreachable_in(baseline_ruleset) 

802 ) 

803 all_unreachable: list[RuleReachability] = RuleReachability.unreachable_in(ruleset) 

804 new_unreachable = [ 

805 result for result in all_unreachable if result.term not in baseline_unreachable_terms 

806 ] 

807 warnings = list(warnings) + [ 

808 f"pre-existing unreachable rule (not introduced by this change, not build-breaking here): " 

809 f"{result.term}" 

810 for result in all_unreachable 

811 if result.term in baseline_unreachable_terms 

812 ] 

813 

814 return cls( 

815 errors=errors, 

816 warnings=warnings, 

817 missing_when_problems=[], 

818 so_parent_drift=[], 

819 unreachable_rules=new_unreachable, 

820 contradictory_pairs=[], 

821 redundant_pairs=[], 

822 )

851# -------------------------------------------------------------------------- 

852# Tier 3: whole-ruleset SAT queries via z3 (optional dev dependency). 

853# 

854# Same two query shapes as Tier 2 (reachability, subsumption-edge soundness), same axiom set, but 

855# posed to a solver instead of enumerated -- so a query is not limited to one rule/pair's bounded 

856# atom support the way Tier 2's brute force is. z3 is imported lazily, only when this function is 

857# actually called (--solver=z3), so it stays an optional dependency 

858# ([project.optional-dependencies] validation = ["z3-solver"] in pyproject.toml) rather than a 

859# hard one. 

860# -------------------------------------------------------------------------- 

861 

862 

863class SolverChecks: 

864 """Run whole-rule-set validation through z3.""" 

865 

866 @classmethod 

867 def findings(cls, ruleset: FactRuleSet) -> list[str]: 

868 """Run the whole-ruleset reachability and subsumption-soundness queries via z3. 

869 

870 Args: 

871 ruleset (FactRuleSet): The rule set to check. 

872 

873 Returns: 

874 list[str]: Findings, prefixed ``"UNREACHABLE (z3): ..."`` (a rule whose formula is 

875 unsatisfiable under the axioms -- a real defect) or ``"NOT ENTAILED (z3): ..."`` (a 

876 declared ``subsumes`` edge not backed by formula entailment -- informational, see 

877 RulePairFinding.redundant_in's docstring for why that is not automatically a bug). 

878 

879 Raises: 

880 RuntimeError: If ``z3`` is not installed. Install the optional ``validation`` extra 

881 (``pip install '.[validation]'`` / ``uv sync --extra validation``) to use this. 

882 """ 

883 try: 

884 import z3 # pylint: disable=import-outside-toplevel 

885 except ImportError as import_error: 

886 raise RuntimeError( 

887 "z3-solver is not installed. Install the optional 'validation' extra " 

888 "(e.g. `uv sync --extra validation`) to use --solver=z3." 

889 ) from import_error 

890 

891 atom_names_in_ruleset = sorted({atom_name for rule in ruleset.rules for atom_name in rule.atoms()}) 

892 z3_atoms = {atom_name: z3.Bool(atom_name) for atom_name in atom_names_in_ruleset} 

893 

894 axioms: list[Any] = [] 

895 for atom_name in atom_names_in_ruleset: 

896 registered_fact = REGISTRY.get(atom_name) 

897 if registered_fact is None: 

898 continue 

899 for required_atom in registered_fact.requires: 

900 if required_atom in z3_atoms: 

901 axioms.append(z3.Implies(z3_atoms[atom_name], z3_atoms[required_atom])) 

902 for atom_a, atom_b in _EXCLUSIVITY_AXIOM_PAIRS: 

903 if atom_a in z3_atoms and atom_b in z3_atoms: 

904 axioms.append(z3.Not(z3.And(z3_atoms[atom_a], z3_atoms[atom_b]))) 

905 for atom_a, atom_b in _EQUIVALENCE_AXIOM_PAIRS: 

906 if atom_a in z3_atoms and atom_b in z3_atoms: 906 ↛ 905line 906 didn't jump to line 905 because the condition on line 906 was always true

907 axioms.append(z3_atoms[atom_a] == z3_atoms[atom_b]) 

908 

909 findings: list[str] = [] 

910 

911 for rule in ruleset.rules: 

912 solver = z3.Solver() 

913 solver.add(*axioms) 

914 solver.add(cls._expression_to_z3(rule.when, z3_atoms, z3)) 

915 if solver.check() == z3.unsat: 

916 findings.append(f"UNREACHABLE (z3): {rule.term} is unsatisfiable under the declared axioms") 

917 

918 for rule in ruleset.rules: 

919 for subsumed_term in rule.subsumes: 

920 subsumed_rule = ruleset.by_term.get(subsumed_term) 

921 if subsumed_rule is None: 

922 continue 

923 solver = z3.Solver() 

924 solver.add(*axioms) 

925 solver.add(cls._expression_to_z3(rule.when, z3_atoms, z3)) 

926 solver.add(z3.Not(cls._expression_to_z3(subsumed_rule.when, z3_atoms, z3))) 

927 if solver.check() == z3.sat: 

928 findings.append( 

929 f"NOT ENTAILED (z3): {rule.term} subsumes {subsumed_term} but {rule.term}'s when " 

930 f"does not entail {subsumed_term}'s when (informational)" 

931 ) 

932 return findings

934 @classmethod 

935 def _expression_to_z3(cls, expression: Expr, atoms: dict[str, Any], z3_module: Any) -> Any: 

936 """Translate an Expr tree into a z3 boolean expression. 

937 

938 Typed with ``Any`` throughout rather than a z3 stub: z3 is an optional dependency (see the 

939 module docstring), so this module must import cleanly whether or not it -- and therefore any 

940 type stub for it -- is installed. 

941 

942 Args: 

943 expression (Expr): The formula to translate. 

944 atoms (dict[str, Any]): Fact atom name -> ``z3.BoolRef`` lookup table. 

945 z3_module (Any): The imported ``z3`` module (passed through rather than imported here 

946 again, since the caller already did the lazy import). 

947 

948 Returns: 

949 Any: The equivalent ``z3.BoolRef``. 

950 

951 Raises: 

952 TypeError: If ``expression`` is a node type this translator does not support. 

953 """ 

954 if isinstance(expression, Atom): 

955 return atoms[expression.name] 

956 if isinstance(expression, Not): 

957 return z3_module.Not(cls._expression_to_z3(expression.child, atoms, z3_module)) 

958 if isinstance(expression, And): 

959 return z3_module.And( 

960 *(cls._expression_to_z3(child, atoms, z3_module) for child in expression.children) 

961 ) 

962 if isinstance(expression, Or): 

963 return z3_module.Or( 

964 *(cls._expression_to_z3(child, atoms, z3_module) for child in expression.children) 

965 ) 

966 raise TypeError(f"unsupported expression node type: {type(expression).__name__}")
