43@define(frozen=True, kw_only=True, auto_attribs=True) 

44class CorpusConfig(ConfigSection): 

45 """Which registered corpus to build, and how. 

46 

47 Consumed by ensembl.vep.corpus. Seeds are per stream rather than one global seed because the 

48 generator draws the chromosome sequence, the boundary variants and the fuzz variants from 

49 independent ``random.Random`` instances. 

50 

51 Attributes: 

52 kind (str): Registered corpus name. 

53 parameters (tuple[str, ...]): Repeatable factory-owned ``KEY=VALUE`` parameters. JSON 

54 scalar values are decoded; other values remain strings. 

55 fuzz (int): Number of random variants to add to the default corpus. 

56 step (int): Position increment for the heavy-locus variant sweep. 

57 genome_seed (int): Seed for the default corpus's chromosome sequence and transcript layout. 

58 boundary_seed (int): Seed for the adversarial boundary variants. 

59 fuzz_seed (int): Seed for the random fuzz variants. 

60 heavy_genome_seed (int): Seed for the heavy locus's chromosome. Separate from 

61 ``genome_seed`` because the two corpora have always started from different seeds. 

62 sweep_seed (int): Seed for the heavy-locus variant sweep. 

63 corpus_dir (Path | None): Directory to load a saved corpus from, or write one to. 

64 """ 

65 

66 section_name: ClassVar[str] = "corpus" 

67 

68 kind: str = field( 

69 default="synthetic.all", metadata=option("registered corpus to use", choices=_corpus_kinds) 

70 ) 

71 parameters: tuple[str, ...] = field( 

72 factory=tuple, 

73 metadata=option( 

74 "factory parameter as KEY=VALUE (repeatable)", 

75 flag="corpus_parameter", 

76 metavar="KEY=VALUE", 

77 repeatable=True, 

78 ), 

79 ) 

80 fuzz: int = field(default=4000, metadata=option("fuzz variant count", metavar="N")) 

81 step: int = field(default=13, metadata=option("heavy-locus variant sweep step", metavar="N")) 

82 genome_seed: int = field(default=7, metadata=option("seed for the synthetic genome")) 

83 boundary_seed: int = field(default=11, metadata=option("seed for the boundary variants")) 

84 fuzz_seed: int = field(default=23, metadata=option("seed for the fuzz variants")) 

85 heavy_genome_seed: int = field(default=31, metadata=option("seed for the heavy-locus genome")) 

86 sweep_seed: int = field(default=41, metadata=option("seed for the heavy-locus variant sweep")) 

87 corpus_dir: Path | None = field(default=None, metadata=option("directory to load or save a corpus"))
